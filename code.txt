#include "BluetoothSerial.h"

BluetoothSerial SerialBT;

#define FLOW_SENSOR_PIN 27

volatile unsigned long pulseCount = 0;

unsigned long lastTime = 0;
float flowRate = 0.0;
float totalLitres = 0.0;

void IRAM_ATTR countPulse() {
  pulseCount++;
}

void setup() {
  Serial.begin(115200);

  // Start Bluetooth
  SerialBT.begin("WaterFlow_ESP32");

  // Flow sensor
  pinMode(FLOW_SENSOR_PIN, INPUT_PULLUP);

  attachInterrupt(
    digitalPinToInterrupt(FLOW_SENSOR_PIN),
    countPulse,
    FALLING
  );

  Serial.println("Water Flow System Started");
  Serial.println("Bluetooth: WaterFlow_ESP32");

  SerialBT.println("Water Flow System Started");
}

void loop() {

  if (millis() - lastTime >= 1000) {

    noInterrupts();
    unsigned long pulses = pulseCount;
    pulseCount = 0;
    interrupts();

    // Common YF-S201 approximation:
    // approximately 450 pulses per litre
    flowRate = (pulses / 450.0) * 60.0;

    // Calculate litres used during this 1-second interval
    totalLitres += pulses / 450.0;

    Serial.print("Flow Rate: ");
    Serial.print(flowRate, 2);
    Serial.println(" L/min");

    Serial.print("Total: ");
    Serial.print(totalLitres, 3);
    Serial.println(" L");

    // Send to Bluetooth
    SerialBT.print("Flow Rate: ");
    SerialBT.print(flowRate, 2);
    SerialBT.println(" L/min");

    SerialBT.print("Total: ");
    SerialBT.print(totalLitres, 3);
    SerialBT.println(" L");

    SerialBT.println("----------------");

    lastTime = millis();
  }
}
